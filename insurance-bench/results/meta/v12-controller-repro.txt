const definitions=[['01-sol-high-dev','gpt-6.1-sol-high','dev',2],['02-luna-high-dev','gpt-6-luna-high','dev',.5]];

const supplementaryReviews=['insurance-bench/results/meta/dependency-allocation-runner-independent-review-v0.12.json','insurance-bench/results/meta/v12-independent-opus-review.json'];

const parentRegistrationPath='insurance-bench/results/meta/v11-registration.json';
const validateParent=()=>{
 const a=json(parentAuditPath),r=json(parentRegistrationPath);
 if(a.disposition!=='approve-engineering-only'||a.status!=='final'||a.registrationSha256!==sha(fs.readFileSync(parentRegistrationPath))||a.sourceFingerprint!==r.sourceFingerprint||a.fullCorpusHash!==r.fullCorpusHash||a.issues.length||a.pending.length||a.panels.length!==4||r.panels.length!==4||a.counts.terminalEpisodesAudited!==24)throw new Error('Completed independent v11 findings required before new admission');

};
const validateReviews=()=>Object.fromEntries(supplementaryReviews.map(p=>{const a=json(p);if(a.disposition!=='approve-engineering-only'||a.sourceFingerprint!==sourceFingerprint)throw new Error('Current independent runtime and cross-provider admission required');return[p,sha(fs.readFileSync(p))];}));

 if(fs.existsSync(registrationPath))throw new Error('Registration exists; never overwrite');
 const audit=json(auditPath),l=ledger(),models=json('insurance-bench/models.json'),supplementaryReviewSha256=validateReviews(),parentControlSha256=validateParent();

 const panels=definitions.map(([id,modelAlias,split,forecastUsd])=>{const selected=chosen(String(split)),model=models[String(modelAlias)];if(model?.provider!=='openai')throw new Error('Exact OpenAI model required');return{id,modelAlias,modelConfiguration:model,split,forecastUsd,taskIds:selected.map(t=>t.id),datasetHash:hash(selected),variant:'baseline',repeat:1,limits:{maxTurns:20,maxOutputTokens:16384},promptHash:hash(systemPrompt('baseline')),harnessHash:hash([sourceFingerprint,systemPrompt('baseline'),20,16384]),dispatchReceiptPath:`insurance-bench/results/meta/${slug}-dispatch-${id}.json`};});
 const prospectiveCost=panels.reduce((n,p)=>n+Number(p.forecastUsd),0);
 if(prospectiveCost>75-l.reservedUsd)throw new Error('Full candidate plan unaffordable; no registration or inference');
 const scriptBytes=fs.readFileSync(fileURLToPath(import.meta.url));
 const controllerReview=json(controllerReviewPath);if(controllerReview.disposition!=='approve-engineering-only'||controllerReview.controllerSha256!==sha(scriptBytes))throw new Error('Current independent controller amendment review required');
 if(fs.existsSync(controllerCopy)&&!fs.readFileSync(controllerCopy).equals(scriptBytes))throw new Error('Reproducible controller copy differs');
 fs.writeFileSync(controllerCopy,scriptBytes);
 const reg={registeredAt:new Date().toISOString(),parentControlSha256,perFamily:1,caseOffset:0,supplementaryReviewSha256,status:'prospective-no-inference-before-committed-freeze',generation,sourceFingerprint,sourceFiles,fullCorpusHash:hash(tasks),promptHash:hash(systemPrompt('baseline')),capUsd:75,minimumReservedUsd:l.reservedUsd,ledgerAtRegistration:l,auditPath,auditSha256:sha(fs.readFileSync(auditPath)),planPath,planSha256:sha(fs.readFileSync(planPath)),controllerCopy,controllerSha256:sha(scriptBytes),controllerReviewPath,controllerReviewSha256:sha(fs.readFileSync(controllerReviewPath)),forecastTotalUsd:prospectiveCost,responseFormat:'json_object',panels,guards:['Serial fixed order; no retries, fallback or outcome-selected substitution','Existing canonical shared ledger with totalcap75 and registered floor; never reset','Every remaining panel forecast must fit before dispatch','Source, plan, audit, controller, registration and task/config exact bytes are committed before inference','Native generation and responseFormat plus metadata audited after every terminal run','Material cost overrun/provider/budget/model/process flags pause future calls','Fresh holdouts require separately committed independent development audit and release; no held output access before release','New corpus,18-package limit,repaired runtime and native JSON do not support cross-generation causal hardness claims; One preregistered high-effort tier contrast within exact v12conditions only; three-case screen cannot localize frontier50–60 macro success']};
 fs.writeFileSync(registrationPath,JSON.stringify(reg,null,2)+'\n');console.log(JSON.stringify({prepared:slug,forecastUsd:prospectiveCost,remainingUsd:75-l.reservedUsd,sourceFingerprint}));process.exit(0);
}
const registration=json(registrationPath);
function validate(){
 if(hash(validateParent())!==hash(registration.parentControlSha256)||registration.perFamily!==1||registration.caseOffset!==0||hash(registration.sourceFiles)!==hash(sourceFiles))throw new Error('Parent/control/selection drift');
 if(hash(validateReviews())!==hash(registration.supplementaryReviewSha256))throw new Error("Supplementary independent review drift");
 if(registration.generation!==generation||registration.sourceFingerprint!==sourceFingerprint||registration.fullCorpusHash!==hash(tasks)||registration.promptHash!==hash(systemPrompt('baseline'))||sha(fs.readFileSync(fileURLToPath(import.meta.url)))!==registration.controllerSha256||sha(fs.readFileSync(controllerCopy))!==registration.controllerSha256||sha(fs.readFileSync(planPath))!==registration.planSha256||sha(fs.readFileSync(auditPath))!==registration.auditSha256||sha(fs.readFileSync(controllerReviewPath))!==registration.controllerReviewSha256)throw new Error('Frozen candidate control drift');
 const models=json('insurance-bench/models.json');if(registration.panels.length!==definitions.length)throw new Error('Fixed panel count required');for(let i=0;i<definitions.length;i++){const [id,alias,split,forecast]=definitions[i]!,p=registration.panels[i];if(p.id!==id||p.modelAlias!==alias||p.split!==split||p.forecastUsd!==forecast||hash(p.modelConfiguration)!==hash(models[p.modelAlias])||p.datasetHash!==hash(chosen(p.split))||hash(p.taskIds)!==hash(chosen(p.split).map(t=>t.id))||p.variant!=='baseline'||p.repeat!==1||hash(p.limits)!==hash({maxTurns:20,maxOutputTokens:16384})||p.promptHash!==hash(systemPrompt('baseline'))||p.harnessHash!==hash([sourceFingerprint,systemPrompt('baseline'),20,16384])||p.dispatchReceiptPath!==`insurance-bench/results/meta/${slug}-dispatch-${p.id}.json`)throw new Error('Registered candidate conditions changed');}
 const l=ledger();if(l.reservedUsd<registration.minimumReservedUsd)throw new Error('Shared ledger reset detected');
}
validate();if(mode==='validate'){console.log(JSON.stringify({validated:slug,sourceFingerprint,paidInferencePerformed:false}));process.exit(0);}
const stage=options.get('--stage'),freezeCommit=options.get('--freeze-commit');if(stage!=='dev'||!freezeCommit||!/^[a-f0-9]{40}$/.test(freezeCommit))throw new Error('Exact committed freeze and stage required');
const committed=(p:string)=>{if(path.isAbsolute(p)||p.includes('..'))throw new Error('Relative committed path required');const b=execFileSync('git',['show',`${freezeCommit}:${p}`],{cwd:root});if(!b.equals(fs.readFileSync(p)))throw new Error('Committed candidate control bytes differ');return JSON.parse(b.toString());};
committed(registrationPath);committed(auditPath);committed(controllerReviewPath);for(const p of supplementaryReviews)committed(p);
for(const p of [planPath,controllerCopy,...sourceFiles]){const bytes=execFileSync('git',['show',`${freezeCommit}:${p}`],{cwd:root});if(!bytes.equals(fs.readFileSync(p)))throw new Error('Committed candidate source/control differs');}
const lockPath='.tmp/insurance-bench/candidate-controller.lock',lock=fs.openSync(lockPath,'wx');fs.writeFileSync(lock,JSON.stringify({pid:process.pid,candidate,stage}));process.on('exit',()=>{fs.closeSync(lock);fs.rmSync(lockPath,{force:true});});
const directory=`.tmp/insurance-bench/${slug}-runs`;fs.mkdirSync(directory,{recursive:true});
for(let index=0;index<registration.panels.length;index++){
 const p=registration.panels[index];if(p.split!==stage)continue;validate();if(fs.existsSync(p.dispatchReceiptPath))throw new Error('No duplicate candidate panel or retry');
 for(const prev of registration.panels.slice(0,index)){const r=json(prev.dispatchReceiptPath);if(r.status!=='terminal-native-run'||!r.completeCoverage||r.processFailure||r.providerOrBudgetFailure||r.modelOrEffortMismatch||r.materialSpendOverrun||sha(fs.readFileSync(r.nativeRunPath))!==r.nativeRunSha256)throw new Error('Serial predecessor invalid; pause');}
 const l=ledger(),remainingForecast=registration.panels.slice(index).reduce((n:number,x:any)=>n+x.forecastUsd,0);if(75-l.reservedUsd<remainingForecast)throw new Error('Remaining prospective plan unaffordable');
 const args=['tsx','scripts/insurance-bench-dependency-allocation.ts','run','--model',p.modelAlias,'--split',p.split,'--per-family','1','--case-offset','0','--repeat','1','--variant','baseline','--max-output-tokens','16384','--budget','75','--out',directory,'--admission',registrationPath,'--freeze-commit',freezeCommit,'--panel',p.id];
 const before=new Set(fs.readdirSync(directory)),logPath=`.tmp/insurance-bench/${slug}-${p.id}.log`;
 const receipt:any={...p,panelId:p.id,startedAt:new Date().toISOString(),freezeCommit,sourceFingerprint,responseFormat:'json_object',ledgerBefore:l,remainingForecastUsd:remainingForecast,command:['npx',...args],status:'dispatched',logPath};fs.writeFileSync(p.dispatchReceiptPath,JSON.stringify(receipt,null,2)+'\n');
 const log=fs.createWriteStream(logPath),child=spawn('npx',args,{cwd:root,shell:false,stdio:['ignore','pipe','pipe']});child.stdout.on('data',b=>{log.write(b);process.stdout.write(b);});child.stderr.on('data',b=>{log.write(b);process.stderr.write(b);});let spawnFailure=false;const code=await new Promise<number|null>(done=>{child.once('error',()=>{spawnFailure=true;done(null);});child.once('exit',done);});log.end();Object.assign(receipt,{endedAt:new Date().toISOString(),exitCode:code,processFailure:spawnFailure||code!==0,status:'process-ended-awaiting-native-validation',ledgerAfter:ledger()});fs.writeFileSync(p.dispatchReceiptPath,JSON.stringify(receipt,null,2)+'\n');
 const matches=fs.readdirSync(directory).filter(n=>n.endsWith('.json')&&!before.has(n)).map(name=>({name,bytes:fs.readFileSync(path.join(directory,name))})).map(x=>({...x,run:JSON.parse(x.bytes.toString())})).filter(x=>x.run.datasetHash===p.datasetHash&&hash(x.run.model)===hash(p.modelConfiguration)&&hash(x.run.taskIds)===hash(p.taskIds));if(matches.length!==1)throw new Error('Exactly one native candidate run required');
 const {name,bytes,run}=matches[0]!;if(run.version!==generation||run.responseFormat!=='json_object'||run.harnessHash!==p.harnessHash||run.repeat!==p.repeat||hash(run.limits)!==hash(p.limits))throw new Error('Native candidate configuration mismatch');const nativeRunPath='insurance-bench/results/'+name;if(fs.existsSync(nativeRunPath))throw new Error('Never overwrite native candidate');fs.writeFileSync(nativeRunPath,bytes);
 Object.assign(receipt,{status:'terminal-native-run',nativeRunPath,nativeRunSha256:sha(bytes),actualEstimatedCostUsd:run.estimatedCostUsd,completeCoverage:run.completeCoverage,benchmarkScore:run.benchmarkScore,weightedDiagnosticScore:run.summary.score,observedTruncation:run.episodes.some((e:any)=>e.truncatedOutput||e.trace.some((t:any)=>/(?:max[_ -]?tokens|max[_ -]?output[_ -]?tokens|length|incomplete)/i.test(t.finishReason??''))),modelOrEffortMismatch:run.episodes.some((e:any)=>e.trace.some((t:any)=>t.model!==p.modelConfiguration.model||(t.reasoningEffort??null)!==(p.modelConfiguration.reasoningEffort??null))),providerOrBudgetFailure:run.episodes.some((e:any)=>['provider-error','budget-stopped'].includes(e.status)),materialSpendOverrun:run.estimatedCostUsd>Math.max(p.forecastUsd*1.5,p.forecastUsd+.5)});fs.writeFileSync(p.dispatchReceiptPath,JSON.stringify(receipt,null,2)+'\n');console.log(JSON.stringify({panel:p.id,macro:run.benchmarkScore,diagnostic:run.summary.score,costUsd:run.estimatedCostUsd,remainingUsd:75-receipt.ledgerAfter.reservedUsd}));if(receipt.processFailure||receipt.providerOrBudgetFailure||receipt.modelOrEffortMismatch||receipt.materialSpendOverrun)throw new Error('Native limitation requires review before next call');

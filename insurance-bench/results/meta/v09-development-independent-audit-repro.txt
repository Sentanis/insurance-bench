function independent(documents:any[]){
 const rows=documents.flatMap(d=>d.rows),ref=new Map(rows.map(r=>[r.id,documents.find(d=>d.rows.includes(r))]));
 const within=(r:any,t:string,a='from',b='to')=>Date.parse(r[a])<=Date.parse(t)&&Date.parse(t)<Date.parse(r[b]);
 const scope=(r:any)=>JSON.stringify(r.kind==='authority'?[r.kind,r.personId,r.role,r.subjectId]:r.kind==='alias'?[r.kind,r.subject,r.label]:r.kind==='observation'?[r.kind,r.subjectId,r.predicate]:[r.kind,r.packId,r.activity]);
 const live=(eligible:any[])=>eligible.filter(x=>!eligible.some(y=>scope(x)===scope(y)&&y.supersedes.includes(x.id)));
 const product=(groups:string[][][])=>groups.reduce<string[][]>((a,b)=>a.flatMap(x=>b.map(y=>[...new Set([...x,...y])])),[[]]);
 const credential=(person:string,role:string,subject:string,t:string)=>live(rows.filter(r=>r.kind==='authority'&&r.signed&&r.personId===person&&r.role===role&&(r.subjectId===subject||r.subjectId==='*')&&within(r,t))).map(r=>[r.id]);
 const inv=(covers:string)=>rows.filter(r=>r.kind==='inventory'&&r.signed&&r.complete&&r.covers===covers).map(r=>[r.id]);
 const resolve=(candidates:any[],eligible:any[],value:(r:any)=>any,role:string,subject:(r:any)=>string,t:string,covers:string)=>{
  const active=live(eligible),witness=(r:any)=>credential(r.signerId,role,subject(r),t).map(p=>[r.id,...p]);
  if(!active.length)return{v:null,p:product([inv(covers),inv('authority'),[candidates.map(r=>r.id)]])};
  const distinct=new Set(active.map(r=>JSON.stringify(value(r))));
  if(distinct.size===1)return{v:value(active[0]),p:active.flatMap(witness)};
  return{v:null,p:active.flatMap((a,i)=>active.slice(i+1).filter(b=>value(a)!==value(b)).flatMap(b=>product([witness(a),witness(b)])))};
 };
 const facts:any={},proofs:any={},requests:any[]=[],identities:any[]=[];
 const asset:string[]=[],actor:string[]=[],policy:string[]=[],unknown:string[]=[];
 const aggregate:any={resolvedAssetIds:[],resolvedActorIds:[],currentPolicyVersionIds:[]};
 for(const e of rows.filter(r=>r.kind==='event'&&r.signed).sort((a,b)=>a.eventId.localeCompare(b.eventId))){
  const ids:any={};
  for(const s of ['asset','actor','site']){
   const candidates=rows.filter(r=>r.kind==='alias'&&r.subject===s&&r.label===e[s+'Label']);
   const eligible=candidates.filter(r=>r.signed&&within(r,e.at)&&credential(r.signerId,'alias-registrar',r.entityId??r.label,e.at).length);
   ids[s]=resolve(candidates,eligible,r=>r.entityId,'alias-registrar',r=>r.entityId??r.label,e.at,'aliases');
   const p=ids[s].p.map((v:string[])=>[e.id,...v]);identities.push(p);
   if(ids[s].v===null)unknown.push(e.eventId+':'+s);
   else if(s==='asset')asset.push(e.eventId+':'+ids[s].v);else if(s==='actor')actor.push(e.eventId+':'+ids[s].v);
   if(s==='asset')aggregate.resolvedAssetIds.push(p);if(s==='actor')aggregate.resolvedActorIds.push(p);
  }
  const candidates=rows.filter(r=>r.kind==='policy'&&r.activity===e.activity),eligible=live(candidates.filter(r=>r.signed&&within(r,e.at,'effectiveFrom','effectiveTo')&&credential(r.signerId,'policy-issuer',r.packId,e.at).length));
  const revision=Math.max(...eligible.map(r=>r.revision)),selected=eligible.filter(r=>r.revision===revision);
  const packs=new Set(selected.map(r=>r.packId));if(packs.size>1)throw Error('Interpreter current one-pack scope exceeded');
  const chosen=selected.length&&new Set(selected.map(r=>JSON.stringify(r.clauses))).size===1?selected[0]:null;
  const policyWitness=(r:any)=>credential(r.signerId,'policy-issuer',r.packId,e.at).map(p=>[e.id,r.id,...p]);
  const pp=chosen?selected.flatMap(policyWitness):selected.length?product(selected.map(policyWitness)):product([[[e.id]],inv('policies'),inv('authority'),[candidates.map(r=>r.id)]]);
  aggregate.currentPolicyVersionIds.push(pp);identities.push(pp);
  if(chosen)policy.push(e.eventId+':'+chosen.packId+'@'+chosen.revision);else unknown.push(e.eventId+':policy');
  const lookup=(subject:any,predicate:string,role:string)=>{
   if(subject===null)return{v:null,p:[[]]};
   const candidates=rows.filter(r=>r.kind==='observation'&&r.subjectId===subject&&r.predicate===predicate);
   return resolve(candidates,candidates.filter(r=>r.signed&&r.firstHand&&within(r,e.at)&&credential(r.signerId,role,subject,e.at).length),r=>r.value,role,()=>subject,e.at,'observations');
  };
  const condition=(c:any):{v:number,p:string[][]}=>{
   if(c.kind==='test'){const id=ids[c.subject],input=lookup(id.v,c.predicate,c.requiredRole);return{v:input.v===null?-1:input.v===c.equals?1:0,p:product([id.p,input.p])};}
   if(c.kind==='not'){const child=condition(c.condition);return{v:child.v===-1?-1:1-child.v,p:child.p};}
   const children=c.conditions.map(condition),decisive=c.kind==='all'?0:1,w=children.filter(x=>x.v===decisive);
   return w.length?{v:decisive,p:w.flatMap(x=>x.p)}:{v:children.some(x=>x.v===-1)?-1:1-decisive,p:product(children.map(x=>x.p))};
  };
  for(let i=1;i<=6;i++){
   const obligation=e.eventId+'.R'+i,c=chosen?.clauses.find(c=>c.code==='R'+i);let status='unresolved',p=pp;
   if(c){const result=condition(c.condition);p=product([pp,result.p]);if(result.v===0)status='not-applicable';else if(result.v===1){const id=ids[c.requestSubject],keeper=lookup(id.v,c.custodianPredicate,c.custodianRole);p=product([p,id.p,keeper.p]);if(typeof id.v==='string'&&typeof keeper.v==='string'&&keeper.v.length){status='applicable';requests.push({obligationId:obligation,eventId:e.eventId,subjectId:id.v,material:c.material,custodianId:keeper.v,periodStart:new Date(Date.parse(e.at)-c.lookbackHours*3600000).toISOString(),periodEnd:new Date(Date.parse(e.at)+c.forwardHours*3600000).toISOString()});}}}
   facts[obligation]=status;proofs[obligation]=p;
  }
 }
 Object.assign(facts,{resolvedAssetIds:asset.sort(),resolvedActorIds:actor.sort(),currentPolicyVersionIds:policy.sort(),unresolvedSubjectIds:unknown.sort()});
 for(const key of Object.keys(aggregate))proofs[key]=product(aggregate[key]);proofs.unresolvedSubjectIds=product(identities);
 const retained:string[]=[],invalidated:string[]=[];

 return{facts,requests,retainedApprovalIds:retained.sort(),invalidatedApprovalIds:invalidated.sort(),proofs,ref};
}

function auditEpisode(p:any,episode:any){
  const task=tasks.find(t=>t.id===episode.taskId);if(!task){issue(p.id,'unexpected-task',episode.taskId);return null;}
  const issuesBefore=issues.length,env=createInvestigationEnvironment(task),failures:any[]=[],packets:any[]=[],toolErrors:any[]=[],protocolProblems:any[]=[];let protocolErrors=0;
  for(const trace of episode.trace){
   if(trace.model!==p.modelConfiguration.model||(trace.reasoningEffort??null)!==(p.modelConfiguration.reasoningEffort??null))issue(p.id+'/'+task.id,'native-model-effort-mismatch',{turn:trace.turn,model:trace.model,effort:trace.reasoningEffort});
   let results:any[];
   try{results=parseActions(trace.response).map(action=>{const before={version:env.state.version,generation:env.state.generation};const result=env.execute(action.tool,action.args);if('error'in result)toolErrors.push({turn:trace.turn,tool:action.tool,before,args:action.args,result});if(action.tool==='save_packet'&&!('error'in result)){
    const gold=independent(env.state.documents),engine=deriveInvestigationDecisions({documents:env.state.documents});
    for(const key of ['facts','requests','retainedApprovalIds','invalidatedApprovalIds'])if(!same((gold as any)[key],(engine as any)[key]))issue(p.id+'/'+task.id,'independent-source-disagreement',{turn:trace.turn,key,independent:(gold as any)[key],engine:(engine as any)[key]});
    const wrong=Object.keys(gold.facts).filter(k=>!semanticValue(env.state.packet!.facts[k],gold.facts[k]));
    const unsupported=Object.keys(gold.facts).filter(k=>!gold.proofs[k].some((proof:string[])=>proof.every(id=>env.state.packet!.evidence[k].some(ref=>ref.rowId===id))));
    packets.push({turn:trace.turn,generation:env.state.generation,packetRevision:env.state.packet!.revision,wrongFactFields:wrong,unsupportedEvidenceFields:unsupported});
   }return{tool:action.tool,result};});}
   catch(error){protocolErrors++;protocolProblems.push({turn:trace.turn,finishReason:trace.finishReason,usage:trace.usage,parseError:error instanceof Error?error.message:String(error),responseTail:trace.response.slice(-160)});results=[{error:'Invalid JSON action protocol; return the declared actions array and recover',outputLimited:/(?:max[_ -]?tokens|max[_ -]?output[_ -]?tokens|length|incomplete)/i.test(trace.finishReason??'')}];}
   if(!same(results,trace.results))issue(p.id+'/'+task.id,'replayed-tool-results-mismatch',{turn:trace.turn});
  }
  if(!same(env.state,episode.state))issue(p.id+'/'+task.id,'replayed-state-mismatch',null);
  if(protocolErrors!==episode.interfaceErrors)issue(p.id+'/'+task.id,'interface-error-count-mismatch',{protocolErrors,native:episode.interfaceErrors});
  const observedTruncation=episode.trace.some(t=>/(?:max[_ -]?tokens|max[_ -]?output[_ -]?tokens|length|incomplete)/i.test(t.finishReason??''));
  if(observedTruncation!==episode.truncatedOutput)issue(p.id+'/'+task.id,'truncation-flag-mismatch',{observedTruncation,native:episode.truncatedOutput});
  const tracedCost=episode.trace.reduce((sum,t)=>sum+(t.usage.input*p.modelConfiguration.inputUsdPerMillion+t.usage.output*p.modelConfiguration.outputUsdPerMillion)/1e6,0);
  if(Math.abs(tracedCost-episode.estimatedCostUsd)>1e-9)issue(p.id+'/'+task.id,'trace-cost-mismatch',{tracedCost,native:episode.estimatedCostUsd});
  const score=gradeInvestigationTask(task,env.state);if(!env.finished)score.allPass=false;
  if(episode.score&&!same(score,episode.score))issue(p.id+'/'+task.id,'replayed-score-mismatch',{replayed:score,native:episode.score});
  const gold=independent(env.state.documents),engine=deriveInvestigationDecisions({documents:env.state.documents});
  for(const k of ['facts','requests','retainedApprovalIds','invalidatedApprovalIds'])if(!same((gold as any)[k],(engine as any)[k]))issue(p.id+'/'+task.id,'final-independent-source-disagreement',k);
  if(env.state.generation===3){const compact=['E1','E2'].flatMap(e=>Array.from({length:6},(_,i)=>({applicable:'A','not-applicable':'N',unresolved:'U'} as any)[gold.facts[e+'.R'+(i+1)]]));if(!same(compact,literalFinal[task.investigation.mechanic]))issue(p.id+'/'+task.id,'hand-calculated-final-disagreement',{compact});}
  if(episode.score){
   const result=new Map(episode.score.criteria.map(c=>[c.id,c.passed]));
   for(const key of Object.keys(gold.facts)){
    const correct=Boolean(env.state.packet&&Object.hasOwn(env.state.packet.facts,key)&&semanticValue(env.state.packet.facts[key],gold.facts[key]));
    const refs=env.state.packet?.evidence[key]??[],current=refs.length>0&&refs.every(r=>env.state.readRefs.includes(`${r.documentId}@${r.version}:${r.rowId}`)&&env.state.documents.some(d=>d.id===r.documentId&&d.version===r.version&&d.rows.some(row=>row.id===r.rowId)));
    const sufficient=current&&gold.proofs[key].some((proof:string[])=>proof.every(id=>refs.some(r=>r.rowId===id)));
    if(result.get('fact:'+key)!==correct)issue(p.id+'/'+task.id,'native-fact-false-grade',{key,independent:correct,native:result.get('fact:'+key)});
    if(result.get('evidence:'+key)!==(correct&&sufficient))issue(p.id+'/'+task.id,'native-evidence-false-grade',{key,independent:correct&&sufficient,native:result.get('evidence:'+key)});
   }
   const currentRequests=env.state.business.requestGeneration===3&&uniqueSetSame(env.state.requests.map(scopedRequest),gold.requests.map(scopedRequest))&&uniqueSetSame(env.state.business.retainedApprovalIds,gold.retainedApprovalIds)&&uniqueSetSame(env.state.business.invalidatedApprovalIds,gold.invalidatedApprovalIds);
   if(result.get('workflow:currentRequests')!==currentRequests)issue(p.id+'/'+task.id,'native-request-false-grade',{independent:currentRequests,native:result.get('workflow:currentRequests')});
  }
  if(episode.score)for(const c of episode.score.criteria.filter(c=>!c.passed))failures.push(c);
  const flags={truncatedOutput:episode.truncatedOutput,finishReasons:episode.trace.map(t=>t.finishReason),terminalStatus:episode.status,providerOrBudgetFailure:['provider-error','budget-stopped'].includes(episode.status),interfaceErrors:protocolErrors};
  if(!same(investigationGoals(task,env.state),episode.actualGoals))issue(p.id+'/'+task.id,'actual-goal-mismatch',null);
  return {taskId:task.id,status:episode.status,traceSha256:crypto.createHash('sha256').update(JSON.stringify(episode.trace)).digest('hex'),score:episode.score?.score,allPass:episode.score?.allPass,criticalFailure:episode.score?.criticalFailure,actualGoals:episode.actualGoals,turns:episode.trace.length,flags,packets,toolErrors,protocolProblems,failedCriteria:failures,fullCurrentSourceFacts:gold.facts,fullCurrentSourceRequests:gold.requests,approvalClassifications:{retained:gold.retainedApprovalIds,invalidated:gold.invalidatedApprovalIds},sourceConsistency:issues.length===issuesBefore};
}

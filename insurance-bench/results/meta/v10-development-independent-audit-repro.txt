const AND=(...xs:number[])=>xs.includes(0)?0:xs.every(x=>x===1)?1:-1;
const OR=(...xs:number[])=>xs.includes(1)?1:xs.every(x=>x===0)?0:-1;
const NOT=(x:number)=>x===-1?-1:1-x;
function hand(task:any,index:number,generation:number,response:boolean){
 const v=patterns[index%4]!,eligible=response?v.e:-1,hold=response?1:-1;
 const p=generation===1?1:0,d=generation===1?1:v.cd,w=generation===1?0:v.cw,r=generation===1?0:v.cr,t=generation===1?0:v.ct;
 const assigned=OR(p,AND(d,v.ack)),blocked=AND(hold,NOT(w));
 const first=[AND(eligible,task.family==='evidence-brief'?AND(d,v.ack):assigned,NOT(blocked)),assigned,task.family==='fraud-referral'?AND(blocked,OR(t,NOT(p))):blocked,AND(eligible,OR(NOT(hold),w)),AND(d,NOT(r)),OR(v.preserve,task.family==='fraud-referral'?AND(eligible,NOT(r),NOT(t)):AND(eligible,NOT(r)))];
 const secondHold=response?(generation===3?0:1):-1;
 const second=[AND(OR(v.preserve,v.bd),NOT(OR(v.br,AND(secondHold,v.alternate)))),AND(v.alternate,secondHold),OR(v.preserve,AND(v.bd,NOT(v.alternate))),generation===3?AND(OR(NOT(secondHold),v.bw),NOT(v.br)):AND(NOT(secondHold),OR(v.bp,v.bw)),AND(v.bp,NOT(v.br)),generation===3?AND(OR(v.preserve,v.bd),NOT(v.br)):OR(AND(secondHold,NOT(v.bw)),AND(v.preserve,NOT(v.br)))];
 const result=(truth:number,subject:number,event:number)=>truth===0?'N':truth===-1?'U':event===1&&((subject===0&&!response)||(subject===1&&generation>1&&!v.cc))?'U':'A';
 return [...first.map((x,i)=>result(x,[0,1,2,0,1,2][i]!,1)),...second.map((x,i)=>result(x,[0,1,2,generation===3?1:0,1,2][i]!,2))];
}

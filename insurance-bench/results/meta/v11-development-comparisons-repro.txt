import fs from 'node:fs';
import crypto from 'node:crypto';
import {compareRuns} from '../../insurance-bench/compare.ts';
import {admitRegisteredReportPanels,type ReportBuilderIO} from '../../insurance-bench/report-admission.ts';
import type {Run} from '../../insurance-bench/runner.ts';
const base='insurance-bench/results/meta/';
const io:ReportBuilderIO={readJson:p=>JSON.parse(fs.readFileSync(p,'utf8')),sha:p=>crypto.createHash('sha256').update(fs.readFileSync(p)).digest('hex'),exists:p=>fs.existsSync(p),listRuns:()=>[]};
const regPath=base+'v11-registration.json',auditPath=base+'v11-development-independent-audit.json';
const reg=io.readJson(regPath) as {sourceFingerprint:string;fullCorpusHash:string;panels:{id:string;dispatchReceiptPath:string}[]};
const rr=admitRegisteredReportPanels(regPath,[auditPath],io);
if(rr.length!==4)throw new Error('Exact four-panel plan required');
const definitions=[{baseline:1,treatment:0,axis:'effort' as const,name:'Sol high minus low'},{baseline:2,treatment:3,axis:'effort' as const,name:'Luna high minus low'},{baseline:3,treatment:0,axis:'tier' as const,name:'Sol high minus Luna high'}];
const contrasts=definitions.map(d=>{
 const baseline=rr[d.baseline]!,treatment=rr[d.treatment]!;
 const receipts=[d.baseline,d.treatment].map(i=>io.readJson(reg.panels[i]!.dispatchReceiptPath) as {observedTruncation?:boolean;providerOrBudgetFailure?:boolean;processFailure?:boolean;modelOrEffortMismatch?:boolean;materialSpendOverrun?:boolean});
 const limited=receipts.some(r=>r.observedTruncation||r.providerOrBudgetFailure||r.processFailure||r.modelOrEffortMismatch||r.materialSpendOverrun)||[baseline,treatment].some(r=>r.episodes.some(e=>(e as typeof e&{truncatedOutput?:boolean}).truncatedOutput||e.trace.some(t=>/(?:max[_ -]?tokens|max[_ -]?output[_ -]?tokens|length|incomplete)/i.test(t.finishReason??''))));
 if([baseline,treatment].some(r=>(r as Run&{headlineMetric?:string}).headlineMetric!=='macro-task-success'))throw new Error('Native headline metric binding failed');
 const macro=(r:Run)=>(r as Run&{benchmarkScore:number|null}).benchmarkScore;
 const baselineMacro=macro(baseline),treatmentMacro=macro(treatment);
 if(limited||typeof baselineMacro!=='number'||typeof treatmentMacro!=='number'||!Number.isFinite(baselineMacro)||!Number.isFinite(treatmentMacro)||baselineMacro<0||baselineMacro>100||treatmentMacro<0||treatmentMacro>100)return {name:d.name,status:'unavailable',reason:'Cumulative limitation or unavailable native headline bars a clean contrast',baselineRunId:baseline.runId,treatmentRunId:treatment.runId};
 return {name:d.name,status:'available',nativeMacroDifference:treatmentMacro-baselineMacro,comparison:compareRuns(baseline,treatment,d.axis)};
});
const result={analyzedAt:new Date().toISOString(),sourceFingerprint:reg.sourceFingerprint,fullCorpusHash:reg.fullCorpusHash,registrationSha256:io.sha(regPath),independentAuditSha256:io.sha(auditPath),contrasts,scope:'Six development cases per model/effort, one attempt, three correlated templated families. Actual completed-task macro and diagnostic partial credit are distinct. No pooled cross-generation ranking, causal hardness effect or professional-validity claim.',rounding:'Paired diagnostic deltas use native per-case values; subtracting rounded summaries may differ by0.01.'};
fs.writeFileSync(base+'v11-development-comparisons.json',JSON.stringify(result,null,2)+'\n');
console.log(JSON.stringify(result));
